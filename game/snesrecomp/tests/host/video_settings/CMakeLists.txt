cmake_minimum_required(VERSION 3.16)
project(snes_video_settings_tests C)
enable_testing()
set(CMAKE_C_STANDARD 11)
set(RECOMP_UI_ROOT "" CACHE PATH "recomp-ui checkout with SNES display-aspect persistence")
add_executable(video_settings_test video_settings_test.c)
target_include_directories(video_settings_test PRIVATE
    ../../../runner/src "${RECOMP_UI_ROOT}/src")
if(MSVC)
    target_compile_options(video_settings_test PRIVATE /UNDEBUG)
else()
    target_compile_options(video_settings_test PRIVATE -UNDEBUG)
    target_link_libraries(video_settings_test PRIVATE m)
endif()
add_test(NAME snes_video_settings COMMAND video_settings_test)
